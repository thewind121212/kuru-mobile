enum ProductStatus {
  active,
  inactive,
  archived;

  String get wire => switch (this) {
    ProductStatus.active => 'ACTIVE',
    ProductStatus.inactive => 'INACTIVE',
    ProductStatus.archived => 'ARCHIVED',
  };

  static ProductStatus fromWire(String? wire) => switch (wire) {
    'ACTIVE' => ProductStatus.active,
    'INACTIVE' => ProductStatus.inactive,
    'ARCHIVED' => ProductStatus.archived,
    _ => ProductStatus.active,
  };
}

@freezed
class ProductSummary with _$ProductSummary {

    required String id,
    required String name,
    String? imageUrl,
    required ProductStatus status,
    required String baseUnitCode,
    required num sellPricePerUnit,
    required num currentStock,
    required num demandStock,

    required int variantCount,

      name: json['name'] as String,
      imageUrl: (raw == null || raw.isEmpty) ? null : raw,

      brandId: json['brandId'] as String?,

      variantCount: (json['variantCount'] as int?) ?? 0,

// lib/core/network/json_optional.dart
/// Sentinel that distinguishes "send `null` to clear" from "omit key entirely".
///
/// Used by PATCH request bodies whose BE DTO accepts `null` for some fields
/// (to disconnect a relation) but treats an absent key as "leave unchanged".
class JsonOptional<T> {
  const JsonOptional.set(T this.value) : isSet = true;
  const JsonOptional.clear() : value = null, isSet = true;

  final T? value;
  final bool isSet;

  void writeTo(Map<String, dynamic> map, String key) {
    map[key] = value;
  }

  static void writeIfPresent<T>(
    Map<String, dynamic> map,
    String key,
    JsonOptional<T>? opt,
  ) {
    opt?.writeTo(map, key);
  }
}
```

class CreateProductBody {
  const CreateProductBody({
    required this.name,
    required this.baseUnitCode,
    required this.sellPrice,
    this.categoryId,
    this.brandId,
    this.description,
    this.importPrice,
    this.exportPrice,
    this.demandStock,
    this.status = 'ACTIVE',
  });

  final String name;
  final String baseUnitCode;
  final num sellPrice;
  final String? categoryId;
  final String? brandId;
  final String? description;
  final num? importPrice;
  final num? exportPrice;
  final num? demandStock;
  final String status;

  Map<String, dynamic> toJson() {
    final m = <String, dynamic>{
      'name': name,
      'status': status,
      'baseUnitCode': baseUnitCode,
      'sellPrice': sellPrice,
    };
    if (categoryId != null) m['categoryId'] = categoryId;
    if (brandId != null) m['brandId'] = brandId;
    if (description != null) m['description'] = description;
    if (importPrice != null) m['importPrice'] = importPrice;
    if (exportPrice != null) m['exportPrice'] = exportPrice;
    if (demandStock != null) m['demandStock'] = demandStock;
    return m;
  }
}

/// Hand-built request maps (per spec §3 — DTO wins over generated client).
class ProductRepository {
  ProductRepository(this._dio);
  final Dio _dio;

  Future<ApiResult<ProductListPage>> getOverview({
    required ProductListFilter filter,
    int page = 1,
    int limit = 50,
  }) async {
    try {
      final q = <String, dynamic>{'page': page, 'limit': limit};
      if (filter.search != null && filter.search!.isNotEmpty) {
        q['searchString'] = filter.search;
      }
      if (filter.categoryId != null) q['categoryIds'] = [filter.categoryId];
      if (filter.brandId != null) q['brandIds'] = [filter.brandId];

      final res = await _dio.get<dynamic>(
        '/product/GetProductOverview',
        queryParameters: q,
      );
      final data = (res.data as Map<String, dynamic>)['data'] as Map<String, dynamic>;
      final items = (data['products'] as List<dynamic>)
          .map((e) => ProductSummary.fromJson(e as Map<String, dynamic>))
          .toList();
      log.i('GetProductOverview ← ${res.statusCode} count=${items.length}');
      return ApiResult.success(ProductListPage(
        items: items,
        page: page,
        limit: limit,
        totalProducts: (data['totalProducts'] as num?)?.toInt() ?? items.length,
      ));
    } on DioException catch (e) {
      log.w('GetProductOverview failed: ${e.message}');
      return ApiResult.failure(mapDioError(e));
    }
  }

  Future<ApiResult<ProductDetail>> getById(String productId) async {
    try {
      final res = await _dio.get<dynamic>(
        '/product/GetProductById',
        queryParameters: {'productId': productId},
      );
      final data = (res.data as Map<String, dynamic>)['data'] as Map<String, dynamic>;
      log.i('GetProductById ← ${res.statusCode} id=${data['id']}');
      return ApiResult.success(ProductDetail.fromJson(data));
    } on DioException catch (e) {
      log.w('GetProductById($productId) failed: ${e.message}');
      return ApiResult.failure(mapDioError(e));
    }
  }

  Future<ApiResult<String>> create(CreateProductBody body) async {
    try {
      final res = await _dio.post<dynamic>(
        '/product/CreateProduct',
        data: body.toJson(),
      );
      final data = (res.data as Map<String, dynamic>)['data'] as Map<String, dynamic>;
      final id = data['productId'] as String?;
      if (id == null) {
        return ApiResult.failure(
          const UnknownException('Empty productId from CreateProduct'),
        );
      }
      log.i('CreateProduct ← ${res.statusCode} id=$id');
      return ApiResult.success(id);
    } on DioException catch (e) {
      log.w('CreateProduct failed: ${e.message}');
      return ApiResult.failure(mapDioError(e));
    }
  }

  Future<ApiResult<void>> updateInfo(UpdateProductInfoBody body) async {
    try {
      final res = await _dio.patch<dynamic>(
        '/product/UpdateProductInfo',
        data: body.toJson(),
      );
      log.i('UpdateProductInfo ← ${res.statusCode} id=${body.productId}');

    } on DioException catch (e) {
      log.w('UpdateProductInfo(${body.productId}) failed: ${e.message}');
      return ApiResult.failure(mapDioError(e));
    }
  }

final productDioProvider = Provider<Dio>((ref) {
  final shared = ref.watch(dioProvider);

  wireSuperTokensToDio(d);

final productRepositoryProvider = Provider<ProductRepository>((ref) {
  return ProductRepository(ref.watch(productDioProvider));

class ProductListNotifier
    extends FamilyAsyncNotifier<ProductListPage, ProductListFilter> {
  static const int _limit = 50;
  int _page = 1;
  List<ProductSummary> _accum = [];

  @override
  Future<ProductListPage> build(ProductListFilter arg) async {
    _page = 1;
    _accum = [];
    final repo = ref.watch(productRepositoryProvider);
    final result = await repo
        .getOverview(filter: arg, page: _page, limit: _limit)
        .unwrap();
    _accum = result.items;
    return result;
  }

  Future<void> loadMore() async {
    final current = state.valueOrNull;
    if (current == null || !current.hasMore) return;
    state = AsyncValue<ProductListPage>.loading().copyWithPrevious(state);
    try {
      final repo = ref.read(productRepositoryProvider);
      final next = await repo
          .getOverview(filter: arg, page: _page + 1, limit: _limit)
          .unwrap();
      _page += 1;
      _accum = [..._accum, ...next.items];
      state = AsyncValue.data(ProductListPage(
        items: _accum,
        page: _page,
        limit: _limit,
        totalProducts: next.totalProducts,
      ));
    } catch (e, st) {
      state = AsyncValue.error(e, st).copyWithPrevious(state);
    }
  }
}

    ProductListNotifier, ProductListPage, ProductListFilter>(
  ProductListNotifier.new,
);

final productByIdProvider =
    FutureProvider.family<ProductDetail, String>((ref, id) async {
  final repo = ref.watch(productRepositoryProvider);
  return repo.getById(id).unwrap();
});

final canWriteProductsProvider = Provider<bool>((ref) {
  final permsAsync = ref.watch(myPermissionsProvider);
  return permsAsync.maybeWhen(
    data: (p) => p.orgPerms.contains('product.write'),
    orElse: () => false,
  );

/// Returns `true` on success, `false` on failure, `null` if user cancelled.
Future<bool?> showProductArchiveDialog(
  BuildContext context, {
  required String productId,
}) {
  return showKConfirmDialog<bool>(
    context: context,
    title: 'Ngừng kinh doanh sản phẩm?',
    body: 'Sản phẩm sẽ bị ẩn khỏi bán hàng và danh mục. '
        'Lịch sử nhập xuất kho và mua hàng được giữ nguyên.',
    confirmLabel: 'Ngừng kinh doanh',
    destructive: true,
    onConfirm: (ctx) async {
      final container = ProviderScope.containerOf(ctx);
      final repo = container.read(productRepositoryProvider);
      final result = await repo.updateInfo(
        UpdateProductInfoBody(productId: productId, status: 'ARCHIVED'),
      );
      switch (result) {
        case ApiSuccess():
          KNotify.success(ctx, 'Đã ngừng kinh doanh sản phẩm');
          container.invalidate(productByIdProvider(productId));
          container.invalidate(productListProvider);
          return true;
        case ApiFailure(:final err):
          KNotify.warning(ctx, err.message);
          return false;
      }
    },
  );
}

### Mirroring web — constants

When porting a web feature, scan `fe/src/components/<module>/constants.ts` (and adjacent) for hardcoded business constants. Web owns lookup lists that BE doesn't expose via endpoint (units, color palette, status labels). Port verbatim into `lib/features/<module>/data/`. Cross-check before assuming a constant comes from BE.

### OpenAPI is transport, not contract

Generated dart-dio in `lib/api/` is the wire transport. The request body and field set are governed by BE Zod DTO (`be/core/domains/<domain>/dto/`), not by the generated request type. When the two disagree, hand-build the request map in the repository — do not pass generated request models blindly. See `lib/features/catalog/products/data/product_repository.dart` for the reference pattern.
```

/// Open the create sheet (initial == null) or the edit sheet (initial == detail).
Future<void> showCreateEditProductSheet(BuildContext context, {ProductDetail? initial}) {
  return showKModalSheet<void>(
    context: context,
    title: initial == null ? 'Sản phẩm mới' : 'Sửa sản phẩm',
    showCancel: true,
    confirmLabel: 'Lưu',
    body: (ctx) => CreateEditProductSheet(initial: initial),
  );
}

class CreateEditProductSheet extends ConsumerStatefulWidget {
  const CreateEditProductSheet({super.key, this.initial});
  final ProductDetail? initial;

  @override
  ConsumerState<CreateEditProductSheet> createState() => _SheetState();
}

class _SheetState extends ConsumerState<CreateEditProductSheet> {
  late final TextEditingController _name;
  late final TextEditingController _desc;
  String? _categoryId;
  String? _brandId;
  String _baseUnitCode = 'each';
  int? _sellPrice;
  String? _nameErr;
  bool _busy = false;

  bool get _isEdit => widget.initial != null;

  @override
  void initState() {
    super.initState();
    final init = widget.initial;
    _name = TextEditingController(text: init?.name ?? '');
    _desc = TextEditingController(text: init?.description ?? '');
    _categoryId = init?.categoryId;
    _brandId = init?.brandId;
    _baseUnitCode = init?.baseUnitCode ?? 'each';
    _sellPrice = init?.sellPrice.toInt();
  }

  @override
  void dispose() {
    _name.dispose();
    _desc.dispose();
    super.dispose();
  }

  bool get _isDirty {
    final init = widget.initial;
    if (init == null) return _name.text.isNotEmpty || _sellPrice != null;
    return _name.text != init.name
      || _desc.text != (init.description ?? '')
      || _categoryId != init.categoryId
      || _brandId != init.brandId
      || _baseUnitCode != init.baseUnitCode
      || _sellPrice != init.sellPrice.toInt();
  }

  bool get _canSave => _name.text.trim().isNotEmpty
    && _sellPrice != null && _sellPrice! >= 0
    && _isDirty && !_busy;

  Future<void> _pickCategory() async {
    final cats = ref.read(categoryOverviewProvider).valueOrNull ?? const [];
    final picked = await showCategoryBrandPickerSheet(
      context: context,
      title: 'Chọn danh mục',
      items: cats.map((c) => PickerItem(id: c.id ?? '', name: c.name ?? '')).toList(),
      selectedId: _categoryId,
    );
    setState(() => _categoryId = picked);
  }

  Future<void> _pickBrand() async {
    final brands = ref.read(brandOverviewProvider).valueOrNull ?? const [];
    final picked = await showCategoryBrandPickerSheet(
      context: context,
      title: 'Chọn thương hiệu',
      items: brands.map((b) => PickerItem(id: b.id ?? '', name: b.name ?? '')).toList(),
      selectedId: _brandId,
    );
    setState(() => _brandId = picked);
  }

  Future<void> _pickUnit() async {
    final picked = await showKActionSheet<String>(
      context: context, title: 'Đơn vị cơ sở',
      items: [
        for (final u in AVAILABLE_UOMS)
          KActionItem(id: u.code, label: u.labelVi, onTap: null),
      ],
    );
    if (picked != null) setState(() => _baseUnitCode = picked);
  }

  Future<void> _submit() async {
    setState(() => _busy = true);
    final repo = ref.read(productRepositoryProvider);
    final ApiResult<dynamic> result;
    if (_isEdit) {
      result = await repo.updateInfo(UpdateProductInfoBody(
        productId: widget.initial!.id,
        name: _name.text == widget.initial!.name ? null : _name.text,
        sellPrice: _sellPrice == widget.initial!.sellPrice.toInt() ? null : _sellPrice,
        baseUnitCode: _baseUnitCode == widget.initial!.baseUnitCode ? null : _baseUnitCode,
        categoryId: _categoryId == widget.initial!.categoryId
            ? null
            : (_categoryId == null
                ? const JsonOptional<String>.clear()
                : JsonOptional.set(_categoryId!)),
        brandId: _brandId == widget.initial!.brandId
            ? null
            : (_brandId == null
                ? const JsonOptional<String>.clear()
                : JsonOptional.set(_brandId!)),
        description: _desc.text == (widget.initial!.description ?? '')
            ? null
            : (_desc.text.isEmpty
                ? const JsonOptional<String>.clear()
                : JsonOptional.set(_desc.text)),
      ));
    } else {
      result = await repo.create(CreateProductBody(
        name: _name.text.trim(),
        baseUnitCode: _baseUnitCode,
        sellPrice: _sellPrice!,
        categoryId: _categoryId,
        brandId: _brandId,
        description: _desc.text.trim().isEmpty ? null : _desc.text.trim(),
      ));
    }

    if (!mounted) return;
    setState(() => _busy = false);

    switch (result) {
      case ApiSuccess(:final data):
        KNotify.success(context, _isEdit ? 'Đã cập nhật' : 'Đã tạo sản phẩm');
        Navigator.of(context).pop();
        ref.invalidate(productListProvider);
        if (_isEdit) {
          ref.invalidate(productByIdProvider(widget.initial!.id));
        } else if (data is String) {
          context.push('/catalog/products/$data');
        }
      case ApiFailure(:final err):
        if (err is BadRequestException) {
          setState(() => _nameErr = err.message);
        } else {
          KNotify.networkError(context, err.message, onRetry: _submit);
        }
    }
  }

  @override
  Widget build(BuildContext context) {
    final cats = ref.watch(categoryOverviewProvider).valueOrNull ?? const [];
    final brands = ref.watch(brandOverviewProvider).valueOrNull ?? const [];
    String? catName;
    if (_categoryId != null) {
      for (final c in cats) {
        if (c.id == _categoryId) { catName = c.name; break; }
      }
    }
    String? brandName;
    if (_brandId != null) {
      for (final b in brands) {
        if (b.id == _brandId) { brandName = b.name; break; }
      }
    }

    return SingleChildScrollView(
      padding: const EdgeInsets.all(16),
      child: Column(crossAxisAlignment: CrossAxisAlignment.stretch, children: [
        KFormField(
          key: const ValueKey('product-name-field'),
          controller: _name,
          label: 'Tên sản phẩm',
          errorText: _nameErr,
          onChanged: (_) => setState(() => _nameErr = null),
        ),
        const SizedBox(height: 12),
        _PickerTrigger(label: 'Danh mục', value: catName ?? 'Tất cả', onTap: _pickCategory),
        const SizedBox(height: 12),
        _PickerTrigger(label: 'Thương hiệu', value: brandName ?? 'Không có', onTap: _pickBrand),
        const SizedBox(height: 12),
        _PickerTrigger(label: 'Đơn vị cơ sở', value: resolveUomLabel(_baseUnitCode), onTap: _pickUnit),
        const SizedBox(height: 12),
        KCurrencyField(
          label: 'Giá bán',
          value: _sellPrice,
          onChanged: (v) => setState(() => _sellPrice = v),
        ),
        const SizedBox(height: 12),
        KFormField(
          controller: _desc, label: 'Mô tả', maxLines: 4,
          onChanged: (_) => setState(() {}),
        ),
        const SizedBox(height: 20),
        FilledButton(
          onPressed: _canSave ? _submit : null,
          child: _busy
            ? const SizedBox(width: 18, height: 18, child: CircularProgressIndicator(strokeWidth: 2))
            : const Text('Lưu'),
        ),
      ]),
    );
  }
}

class _PickerTrigger extends StatelessWidget {
  const _PickerTrigger({required this.label, required this.value, required this.onTap});
  final String label;
  final String value;
  final VoidCallback onTap;
  @override
  Widget build(BuildContext context) {
    return InkWell(
      onTap: onTap,
      borderRadius: BorderRadius.circular(12),
      child: InputDecorator(
        decoration: InputDecoration(
          labelText: label,
          border: const OutlineInputBorder(),
        ),
        child: Row(children: [
          Expanded(child: Text(value, maxLines: 1, overflow: TextOverflow.ellipsis)),
          const Icon(Icons.keyboard_arrow_down),
        ]),
      ),
    );
  }
}
